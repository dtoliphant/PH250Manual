\begin{tikzpicture}
  % Fractional current uncertainty vs shunt resistance for R = 5000 ohm,
  % Delta V_ps = 2 V, 4.88 mV ADC steps and a 1% shunt tolerance
  % (x in kilo-ohms):  N = floor(Vs / 4.88 mV),  f_I = 100 sqrt((1/N)^2 + 0.01^2)
  \begin{scope}[xscale=2,yscale=.3]
    \draw[graph axis] (0,0) -- (5.3,0) node[right,label] {$R_s$ ($\Omega$)};
    \draw[graph axis] (0,0) -- (0,12.8) node[above,label] {$f_I$ (\%)};
    \foreach \x/\t in {0/0,1/1000,2/2000,3/3000,4/4000,5/5000}{\draw[graph tick] (\x,0) -- ++(0,-.25) node[below,tick label] {\t};}
    \foreach \y in {4,8,12}{\draw[graph tick] (0,\y) -- ++(-.04,0) node[left,tick label] {\y};}
    \draw[curve] plot[domain=.016:5,samples=700] (\x,{min(12,100*sqrt((1/floor((2*\x/(5+\x))/0.00488))^2+0.0001))});
    \draw[guide] (.5,0) -- (.5,12);
    \node[label,anchor=west] at (.56,9) {$R_s = 500\,\Omega$};
  \end{scope}
\end{tikzpicture}
