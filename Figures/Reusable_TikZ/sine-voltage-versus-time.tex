\begin{tikzpicture}
  % A sinusoidally varying voltage, V = sin(2 pi t / 6)
  \begin{scope}[xscale=.36,yscale=1.4]
    \foreach \x in {-12,-8,...,12}{\draw[graph grid] (\x,-1.3) -- (\x,1.3);}
    \foreach \y in {-1,-.5,.5,1}{\draw[graph grid] (-12.5,\y) -- (12.5,\y);}
    \draw[graph axis] (-12.8,0) -- (13.2,0) node[right,label] {time};
    \draw[graph axis] (0,-1.45) -- (0,1.55) node[above,label] {voltage};
    \foreach \x in {-12,-8,-4,4,8,12}{\draw[graph tick] (\x,0) -- ++(0,-.06) node[below,tick label] {\x};}
    \foreach \y in {-1,-.5,.5,1}{\draw[graph tick] (0,\y) -- ++(-.2,0) node[left,tick label] {\y};}
    \draw[model] plot[smooth,domain=-12:12,samples=200] (\x,{sin(60*\x)});
  \end{scope}
\end{tikzpicture}
