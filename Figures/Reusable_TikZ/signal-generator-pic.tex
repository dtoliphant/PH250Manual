      \path[fill=otgray!18,rounded corners=.22cm] (-3.90,-1.60) rectangle (4.10,1.40);
      \path[fill=otgray!10,draw=otgray,line width=.95pt,rounded corners=.22cm] (-4.00,-1.50) rectangle (4.00,1.50);

      \foreach \t [count=\i] in {1,10,100,1k,10k,100k}{
        \path[fill=otgray!25,draw=otgray,line width=.45pt,rounded corners=.04cm] (.29+\i*.37,.49) rectangle (.61+\i*.37,.81);

    }
  },
